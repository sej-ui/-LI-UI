<script>
// 고대비/고채도 색상 팔레트 (요청하신 #bada55 느낌의 명도/채도 적용)
const HIGH_CONTRAST_COLORS = [
    '#FF3D00', '#D4E157', '#00E676', '#00B0FF', '#651FFF', 
    '#F50057', '#FFEA00', '#00E5FF', '#76FF03', '#C6FF00',
    '#FF9100', '#FF1744', '#1DE9B6', '#3D5AFE', '#AA00FF'
];

let allData = [];

function goPage(pageId) {
    document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
    document.getElementById(pageId).classList.add('active');
}

function handleFile(e) {
    const file = e.target.files[0];
    const reader = new FileReader();
    reader.onload = function(evt) {
        const data = evt.target.result;
        const workbook = XLSX.read(data, {type: 'binary'});
        const sheetName = workbook.SheetNames[0];
        allData = XLSX.utils.sheet_to_json(workbook.Sheets[sheetName]);
        renderCharts();
    };
    reader.readAsBinaryString(file);
}

function renderCharts() {
    // 1. 데이터 가공 (운임 날짜 기준 & 월 단위 집계)
    const monthlyStats = {}; // { 'YYYY-MM': { '국가': 총액 } }
    const countries = [...new Set(allData.map(d => d['국가']))];
    
    allData.forEach(row => {
        // [운임] 란의 날짜 데이터를 추출 (엑셀 필드명에 맞춰 수정 필요)
        const dateRaw = row['운임날짜'] || row['운임']; 
        if(!dateRaw) return;
        
        const d = new Date(dateRaw);
        if(isNaN(d)) return;
        
        const monthKey = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
        const country = row['국가'];
        const amt = parseFloat(String(row['금액'] || 0).replace(/,/g, ''));
        
        if(!monthlyStats[monthKey]) monthlyStats[monthKey] = {};
        if(!monthlyStats[monthKey][country]) monthlyStats[monthKey][country] = 0;
        monthlyStats[monthKey][country] += amt;
    });

    const sortedMonths = Object.keys(monthlyStats).sort();
    
    // 2. Chart.js 데이터셋 생성
    const datasets = countries.map((country, idx) => ({
        label: country,
        data: sortedMonths.map(m => monthlyStats[m][country] || 0),
        borderColor: HIGH_CONTRAST_COLORS[idx % HIGH_CONTRAST_COLORS.length],
        backgroundColor: HIGH_CONTRAST_COLORS[idx % HIGH_CONTRAST_COLORS.length],
        borderWidth: 3,
        tension: 0.3, // 곡선 처리
        pointRadius: 5 // 월별 점 크기 확대
    }));

    // 3. 그래프 그리기
    const ctx = document.getElementById('mainFreightChart').getContext('2d');
    if(window.myChart) window.myChart.destroy();
    
    window.myChart = new Chart(ctx, {
        type: 'line',
        data: { labels: sortedMonths, datasets: datasets },
        options: {
            responsive: true,
            maintainAspectRatio: false,
            plugins: {
                legend: { position: 'bottom' }
            },
            scales: {
                y: { beginAtZero: true, title: { display: true, text: '금액' } },
                x: { title: { display: true, text: '집계월 (운임 기준)' } }
            }
        }
    });
}
</script>
